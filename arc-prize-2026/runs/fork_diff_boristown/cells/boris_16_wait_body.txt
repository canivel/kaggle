def wait_vllm_ready(timeout=180):
    start = time.time()

    while time.time() - start < timeout:
        try:
            r = requests.get(
                "http://127.0.0.1:1234/v1/models",
                timeout=5
            )

            if r.status_code == 200:
                print("vLLM server ready")
                return True

        except Exception:
            pass

        time.sleep(5)

    raise RuntimeError(
        "vLLM server is not alive before benchmark start"
    )
